262std::shared_ptr<tatami::Matrix<OutputValue_, Index_> > normalize_counts(
263 std::shared_ptr<const tatami::Matrix<InputValue_, Index_> > counts,
264 SizeFactors_ size_factors,
265 const NormalizeCountsOptions& options)
266{
267 auto current_pseudo = options.pseudo_count;
268 if (options.preserve_sparsity && current_pseudo != 1 && options.log) {
269 for (auto& x : size_factors) {
270 x *= current_pseudo;
271 }
272 current_pseudo = 1;
273 }
274
275 static_assert(std::is_floating_point<OutputValue_>::value);
276 if (static_cast<size_t>(size_factors.size()) != static_cast<size_t>(counts->ncol())) {
277 throw std::runtime_error("length of 'size_factors' should be equal to the number of columns of 'counts'");
278 }
279
280
281 if (!options.log) {
282 return std::make_shared<tatami::DelayedUnaryIsometricOperation<OutputValue_, InputValue_, Index_> >(
283 std::move(counts),
284 std::make_shared<tatami::DelayedUnaryIsometricDivideVectorHelper<true, OutputValue_, InputValue_, Index_, SizeFactors_> >(std::move(size_factors), false)
285 );
