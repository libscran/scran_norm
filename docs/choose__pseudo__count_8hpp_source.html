49template<typename Float_>
50Float_ find_quantile(Float_ quantile, std::size_t n, Float_* ptr) {
51 assert(n > 0);
52
53 const std::size_t n_m1 = n - 1;
54 const Float_ raw = n_m1 * quantile;
55 const Float_ upper_index = std::ceil(raw);
56 const Float_ lower_index = std::floor(raw);
57
58 // Sanitize the upper_index in case the conversion of n_m1 to a Float_ is not exact.
59 // It could be Inf, or it could be rounded up, in which case upper_index <= n_m1 might not be true anymore.
60 const std::size_t upper_index_san = std::min(sanisizer::from_float<std::size_t>(upper_index), n_m1);
61
62 std::nth_element(ptr, ptr + upper_index_san, ptr + n);
63 const auto upper = *(ptr + upper_index_san);
64 if (lower_index == upper_index) {
65 return upper;
66 }
67
68 // lower_index < upper_index, so we just take the maximum after the nth_element moved everything below 'upper' to the left.
69 const auto lower = *std::max_element(ptr, ptr + upper_index_san);
70
71 // Equivalent to 'lower * (upper_index - raw) + upper * (raw - lower_index)', given that 'upper_index - lower_index = 1'.
72 return lower + (upper - lower) * (raw - lower_index);
73}

114template<typename Float_>
115Float_ choose_pseudo_count_raw(const std::size_t num, Float_* const size_factors, const ChoosePseudoCountOptions& options) {
116 // Avoid problems with zeros.
117 I<decltype(num)> actual_num = 0;
118 for (I<decltype(num)> i = 0; i < num; ++i) {
119 const auto val = size_factors[i];
120 if (std::isfinite(val) && val > 0) {
121 if (i != actual_num) {
122 size_factors[actual_num] = val;
123 }
124 ++actual_num;
125 }
126 }
127
128 if (actual_num == 0) {
129 return options.min_value;
130 }
131
132 if (options.quantile < 0 || options.quantile > 1) {
133 throw std::runtime_error("'quantile' should be in [0, 1]");
134 }
135 const Float_ lower_sf = internal::find_quantile(options.quantile, actual_num, size_factors);
136 const Float_ upper_sf = internal::find_quantile(1 - options.quantile, actual_num, size_factors);
137
138 // Very confusing formulation in Equation 3, but whatever.
139 const Float_ pseudo_count = (1.0 / lower_sf - 1.0 / upper_sf) / (8 * options.max_bias);
140
141 return std::max(options.min_value, pseudo_count);
142}
